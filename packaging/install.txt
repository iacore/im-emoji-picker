# Install the handwriting build of the emoji picker.
#
# 1. Install the package. It replaces the stock fcitx5-im-emoji-picker-git
#    (pacman asks first); ~/.config/fcitx5, including the Super+E trigger, is
#    not touched.
sudo pacman -U packaging/fcitx5-im-emoji-picker-handwriting-git-*.pkg.tar.zst

# 2. A running fcitx5 keeps the library it mapped at startup, which is the file
#    the upgrade replaced. Restart it, so the new library is the one mapped:
#    the dinit-supervised instance comes back on its own (restart = true).
dinitctl restart fcitx5

# 3. Verify the running instance uses the new library: the output must show
#    fcitx5imemojipicker-handwriting.so and libggml, and no "(deleted)".
for p in $(pgrep -x fcitx5); do echo "pid $p"; ls -l /proc/$p/map_files/ | grep -iE 'imemojipicker|libggml' | sed 's/.*-> //' | sort -u; done

# 4. Super+E opens the picker on the emoji view. Reach the pad by clicking the
#    pen indicator or with Shift+Tab. Write with the mouse or a stylus, then
#    click a candidate or press 1-9 / arrows + Enter. Wipe the pad with a right
#    click, the Clear button, or Ctrl+Backspace.
